  let disposed = false, busy = false, readable = false, phase = '运营中', covered = false;
  let initialChatId, lastKey, pendingReceipt, picked = '', members = {};

  function context() {

  function api() {
    context();
    // Bracket access with an escape keeps the underscore out of the code-block text.

  function controls() {

  function say(text, tone) {

  function el(tag, text, className) {

  // A stable tint per person, so the same member keeps the same face everywhere.
  function hue(name) {
    let sum = 0;
    for (const char of name) sum = (sum + char.codePointAt(0)) % 360;

  }
  function face(name, control) {
    const ring = el('span', undefined, 'face');

  function seat(name, member, gain) {

  function bar(node, value) { node.style.setProperty('--p', Math.max(0, Math.min(100, value)) + '%'); }

  function mark() {
    const member = picked && members[picked];

    for (const button of document.querySelectorAll('[data-pick]')) button.setAttribute('aria-pressed', String(button.dataset.pick === picked));

      const [before, name, after] = copy.pickTo(picked);

  function report(w) {
    ui.last.hidden = !w;
    if (!w) { ui.last.replaceChildren(); return; }

    ledger.append(cell(copy.report.income, copy.yuan(w.income)), cell(copy.report.rent, copy.yuan(w.rent)),
      cell(copy.report.net, copy.signed(net), 'net ' + (net >= 0 ? 'up' : 'down')));
    const roomChips = el('ul', undefined, 'chips');
    for (const room of rooms) {
      if (w.rooms[room] && w.rooms[room] !== '空') roomChips.append(el('li', room + ' · ' + w.rooms[room], w.rooms[room] === '开课' ? 'up' : 'idle'));

  function render(state) {
    phase = state.phase;
    members = Object.fromEntries(Object.entries(state.members).filter(([, member]) => member.status === '在读'));

  // The card's own cover lights the sign; without one the gradient stands alone.
  function cover(ctx) {
    covered = true;
    const card = ctx.characters && ctx.characters[ctx.characterId];
    if (card && card.avatar) ui.hero.style.setProperty('--cover', 'url(' + JSON.stringify('/characters/' + encodeURIComponent(card.avatar)) + ')');
  }

  function refresh() {
    try {
      const ctx = context();
      if (!covered) cover(ctx);
      const raw = ctx.variables.local.get(root);
      const key = typeof raw === 'string' ? raw : JSON.stringify(raw);
      if (key !== lastKey) {
        const state = typeof raw === 'string' && raw !== '' ? JSON.parse(raw) : raw;
        if (!state || typeof state.week !== 'number' || !state.rooms) throw new Error(copy.uninitialized);
        // A change from outside (undo, swipe, the update model) replaces stale feedback.
        if (!busy && lastKey !== undefined) {
          say('');
          ui.details.hidden = true;
        }
        render(state);
        lastKey = key;
      }
      const current = api().lastResult();
      if (pendingReceipt && (!current || !current.ok || !current.receipt || current.receipt.id !== pendingReceipt || current.saved)) pendingReceipt = undefined;
      readable = true;
      ui['read-error'].hidden = true;
    } catch (error) {
      readable = false;
      lastKey = undefined;
      ui['read-error'].textContent = error.message;
      ui['read-error'].hidden = false;
    }
    controls();
  }

  // Rejected preconditions carry the failing comparison; name the shortfall the player can act on.
  function rejection(diagnostic) {
    const condition = diagnostic.condition || {};
    const left = condition.left || {};
    const path = JSON.stringify(left.path || []);
    if (path === JSON.stringify([root, 'phase'])) return copy.need.phase;
    if (path === JSON.stringify([root, 'funds'])) return copy.need.funds;
    if (condition.kind === 'function') return copy.need.done;
    if (condition.operator === 'gt') return copy.need.full;
    if (condition.right && condition.right.value === '在读') return copy.need.gone;
    return '';
  }
  function showFailure(error) {
    const list = error.diagnostics || [{ code: error.code, message: error.message }];
    const first = list[0] || {};
    const reason = copy.errors[first.code] || rejection(first) || first.message || first.code || '';
    say([first.actionTitle ? first.actionTitle + '：' + copy.failed : copy.failed, reason].filter(Boolean).join('\n'), 'bad');
    ui.diagnostic.textContent = JSON.stringify(list, null, 2);
    ui.details.hidden = false;
  }

  function story(result) {
    const changed = name => result.changes.some(item => item.path[0] === root && item.path[1] === name);
    const state = result.state[root];
    // The settled week is already shown in full in the report; feedback only confirms it.
    if (changed('lastWeek') && state.lastWeek) {
      const lines = [copy.closed(state.lastWeek.week)];
      if (result.provenance.some(item => item.source && item.source.kind === 'trigger' && item.source.id === 'first-graduate')) lines.push('第一位会员结业，家里送来 2000 元捐款，口碑上涨。');
      return lines;
    }
    const lines = [];
    for (const name of ['funds', 'reputation', 'exposure', 'waiting']) {
      const item = result.changes.find(change => change.path.length === 2 && change.path[0] === root && change.path[1] === name);
      if (item) lines.push(({ funds: '资金', reputation: '口碑', exposure: '风声', waiting: '待面谈' })[name] + ' ' + item.before + ' → ' + item.after);
    }
    const moved = result.actions.find(action => action.id === 'assign');
    if (moved) lines.unshift(moved.params.member + (moved.params.room ? ' 分到' + moved.params.room : ' 移出了房间'));
    return lines.length ? lines : [copy.unchanged];
  }

  async function run(input) {
    if (disposed || busy || !readable || input && phase !== '运营中') return;
    busy = true;
    controls();
    say(input ? copy.working : copy.saving);
    ui.details.hidden = true;
    try {
      const stateApi = api();
      if (!input) {
        await stateApi.retrySave();
        if (disposed) return;
        context();
        pendingReceipt = undefined;
        say(copy.saved, 'good');
        return;
      }
      const result = await stateApi.apply(input);
      if (disposed) return;
      context();
      if (!result.ok) { showFailure(result); return; }
      pendingReceipt = result.saved === false ? result.receipt.id : undefined;
      say([...story(result), result.saved === false ? copy.unsaved : ''].filter(Boolean).join('\n'), result.saved === false ? 'bad' : 'good');
      if (result.persistence || result.diagnostics.length) {
        ui.diagnostic.textContent = JSON.stringify([...(result.persistence ? [result.persistence] : []), ...result.diagnostics], null, 2);
        ui.details.hidden = false;
      }
    } catch (error) {
      if (!disposed) showFailure(error);
    } finally {
      // Refresh while still busy: this action's own change must not read as an outside change.
      if (!disposed) refresh();
      busy = false;
      controls();
    }
  }

  ui.targets.append(...rooms.map(room => {
    const button = el('button', room);
    button.type = 'button';
    button.dataset.to = room;
    return button;
  }));
  const leave = el('button', copy.leaveRoom, 'quiet');
  leave.type = 'button';
  leave.dataset.to = '';
  const cancel = el('button', copy.cancel, 'quiet');
  cancel.type = 'button';
  cancel.dataset.cancel = '';
  ui.targets.append(leave, cancel);
  ui.legend.replaceChildren(...copy.legend.flatMap(([label, tone, meaning], index) => {
    const name = el('b', label, tone);
    return [index ? ' · ' : '', name, ' ' + meaning];
  }));

  const onClick = event => {
    const target = event.target.closest('[data-act], [data-pick], [data-to], [data-cancel]');
    if (!target || target.disabled) return;
    if (target.dataset.act !== undefined) run([{ action: 'handle', params: { incident: Number(target.dataset.act) } }]);
    else if (target.dataset.pick !== undefined) { picked = picked === target.dataset.pick ? '' : target.dataset.pick; mark(); }
    else if (target.dataset.to !== undefined) {
      const member = picked;
      picked = '';
      mark();
      if (member) run([{ action: 'assign', params: { member, room: target.dataset.to } }]);
    } else { picked = ''; mark(); }
  };
  const onEndWeek = () => run([{ action: 'endweek', params: {} }]);
  const onRetry = () => run(null);
  ui.desk.addEventListener('click', onClick);
  ui.endweek.addEventListener('click', onEndWeek);
  ui.retry.addEventListener('click', onRetry);
  try { initialChatId = host.SillyTavern.getContext().chatId; } catch (error) { showFailure(error); }
  mark();
  refresh();
  const timer = window.setInterval(refresh, 500);
  window.addEventListener('pagehide', () => {
    disposed = true;
    window.clearInterval(timer);
    ui.desk.removeEventListener('click', onClick);
    ui.endweek.removeEventListener('click', onEndWeek);
    ui.retry.removeEventListener('click', onRetry);
  }, { once: true });
})();
